FYI, here is a modifying repro that crashes on M146 without this with fix and when DCHECK(CanElideWriteBarrier(object, value)); is commented out (for context: this is a broken DCHECK that has a side-effects that fixes this bug, but in release builds the DCHECK doesn't run and so this bug does exist):

// Flags: --allow-natives-syntax --verify-heap --maglev

const non_smi_int32 = 0x40000000;
let cst = 52;

let o1 = { x : 17 };
o1.x = 25;

let o2 = { y : "abc" };
o2.y = [];

function foo(c) {
  let phi = 42;
  for (let i = 0; i < 10; i++) {
    phi++;
  }
  cst = phi;
  o1.x = phi;
  let phi2 = c ? phi : non_smi_int32;
  // Requires HeapNumber
  o2.y = phi2;
}

%PrepareFunctionForOptimization(foo);
foo(false);
foo(false);

%OptimizeMaglevOnNextCall(foo);
foo(false);
assertOptimized(foo);

assertEquals(52, cst);
assertEquals(52, o1.x);
assertEquals(non_smi_int32, o2.y);

foo(true);
assertEquals(52, cst);
assertEquals(52, o1.x);
assertEquals(52, o2.y);
This one just crashes because we store a HeapObject in a Smi field, but we don't omit the write barrier. However, I'm reasonably confident that this could be changed in a way that the write barrier ends up being omitted.

And here is a repro where we do forget a write barrier (still DCHECK(CanElideWriteBarrier(object, value)); commented out):

// Flags: --allow-natives-syntax --verify-heap --maglev --no-maglev-loop-peeling

const max_smi = 0x3fffffff;
const non_smi_int32 = max_smi + 1;
let cst = max_smi-1;

let o1 = { x : 17 };
o1.x = 25;

let o2 = { y : "abc" };
o2.y = [];

function foo(c, n) {
  let phi = 0x3ffffffc;

  for (let i = 0; i < n; i++) {
    phi++;
  }

  if (c) {
    cst = phi;
    o1.x = phi;
  } else {
    let p2 = c ? phi : non_smi_int32;
    o2.y = p2;
  }
}

%PrepareFunctionForOptimization(foo);
foo(true, 3);
foo(false, 3);

%OptimizeMaglevOnNextCall(foo);
foo(true, 3);
foo(false, 3);
And so since we do indeed end up forgetting a write barrier, I'm confirming that this needs to be backmerged to 146 & 147 because it could be exploited like https://crbug.com/491410818.
